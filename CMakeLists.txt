cmake_minimum_required(VERSION 3.20)
project(dsee_compat VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
include(GNUInstallDirs)
function(dsee_options target)
  target_compile_definitions(${target} PRIVATE NOMINMAX)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /fp:strict /utf-8)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()
option(DSEE_UNICORN_PROBE_ONLY "Build only the optional Unicorn diagnostic" OFF)
option(DSEE_DYNARMIC_PROBE_ONLY "Build only the optional Dynarmic diagnostic" OFF)
option(DSEE_EXPERIMENTAL_LINUX "Enable the experimental native AArch64 Linux runtime" OFF)
set(DSEE_BACKEND AUTO CACHE STRING "Execution backend: AUTO, NATIVE or DYNARMIC")
set_property(CACHE DSEE_BACKEND PROPERTY STRINGS AUTO NATIVE DYNARMIC)
if(NOT DSEE_BACKEND MATCHES "^(AUTO|NATIVE|DYNARMIC)$")
  message(FATAL_ERROR "DSEE_BACKEND must be AUTO, NATIVE or DYNARMIC")
endif()
if(DSEE_UNICORN_PROBE_ONLY AND DSEE_DYNARMIC_PROBE_ONLY)
  message(FATAL_ERROR "Select only one probe engine per build directory")
endif()
set(dsee_backend ${DSEE_BACKEND})
if(DSEE_UNICORN_PROBE_ONLY)
  set(dsee_backend UNICORN)
elseif(DSEE_DYNARMIC_PROBE_ONLY)
  set(dsee_backend DYNARMIC)
elseif(dsee_backend STREQUAL "AUTO")
  if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 8 AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|X86_64)$")
    set(dsee_backend DYNARMIC)
  else()
    set(dsee_backend NATIVE)
  endif()
endif()
if(dsee_backend STREQUAL "NATIVE")
  if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$" OR
     NOT (APPLE OR (CMAKE_SYSTEM_NAME STREQUAL "Linux" AND DSEE_EXPERIMENTAL_LINUX)))
    message(FATAL_ERROR "NATIVE requires Apple Silicon or experimental AArch64 Linux")
  endif()
elseif(dsee_backend STREQUAL "DYNARMIC" AND NOT DSEE_DYNARMIC_PROBE_ONLY)
  if(NOT (WIN32 AND MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8 AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|X86_64)$"))
    message(FATAL_ERROR "The integrated Dynarmic runtime requires Windows x86-64 with an MSVC-compatible compiler")
  endif()
endif()
message(STATUS "DSEE execution backend: ${dsee_backend}")
add_library(dsee_support STATIC src/elf.cpp src/resources.cpp src/platform.cpp)
target_include_directories(dsee_support PRIVATE src)
dsee_options(dsee_support)
set(DSEE_CONFIG_DEPENDENCIES "")
if(NOT dsee_backend STREQUAL "NATIVE")
  target_compile_definitions(dsee_support PRIVATE DSEE_ELF_PARSER_ONLY)
endif()
if(WIN32)
  target_link_libraries(dsee_support PRIVATE bcrypt)
elseif(NOT APPLE)
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  target_link_libraries(dsee_support PRIVATE OpenSSL::Crypto)
  string(APPEND DSEE_CONFIG_DEPENDENCIES "find_dependency(OpenSSL COMPONENTS Crypto)\n")
endif()
if(NOT dsee_backend STREQUAL "NATIVE")
  if(dsee_backend STREQUAL "DYNARMIC")
    # Validated source revision: a41c380246d3d9f9874f0f792d234dc0cc17c180.
    find_package(dynarmic 6.7.0 EXACT CONFIG REQUIRED)
    set(cpu_source src/cpu_dynarmic.cpp)
    string(APPEND DSEE_CONFIG_DEPENDENCIES "find_dependency(dynarmic 6.7.0 EXACT CONFIG)\n")
  else()
    find_path(UNICORN_INCLUDE_DIR unicorn/unicorn.h REQUIRED)
    find_library(UNICORN_LIBRARY NAMES unicorn unicorn.2 REQUIRED)
    set(cpu_source tools/cpu_unicorn.cpp)
  endif()
  add_library(dsee_emulation STATIC src/emulation.cpp ${cpu_source})
  target_include_directories(dsee_emulation PRIVATE src)
  target_link_libraries(dsee_emulation PRIVATE dsee_support)
  dsee_options(dsee_emulation)
  if(NOT MSVC)
    target_compile_options(dsee_emulation PRIVATE -ffp-contract=off)
  endif()
  if(dsee_backend STREQUAL "DYNARMIC")
    target_link_libraries(dsee_emulation PRIVATE dynarmic::dynarmic)
    target_compile_features(dsee_emulation PRIVATE cxx_std_20)
    target_compile_definitions(dsee_emulation PRIVATE DSEE_DYNARMIC_VERSION="${dynarmic_VERSION}")
  else()
    target_include_directories(dsee_emulation SYSTEM PRIVATE ${UNICORN_INCLUDE_DIR})
    target_link_libraries(dsee_emulation PRIVATE ${UNICORN_LIBRARY})
  endif()
  if(DSEE_UNICORN_PROBE_ONLY OR DSEE_DYNARMIC_PROBE_ONLY OR BUILD_TESTING)
    string(TOLOWER ${dsee_backend} probe_engine)
    add_executable(dsee-${probe_engine}-probe tools/emulation_probe.cpp)
    target_include_directories(dsee-${probe_engine}-probe PRIVATE src)
    target_link_libraries(dsee-${probe_engine}-probe PRIVATE dsee_emulation)
    dsee_options(dsee-${probe_engine}-probe)
    if(NOT MSVC)
      target_compile_options(dsee-${probe_engine}-probe PRIVATE -ffp-contract=off)
    endif()
    if(BUILD_TESTING)
      add_test(NAME ${probe_engine}_without_assets COMMAND dsee-${probe_engine}-probe --self-test)
    endif()
  endif()
endif()
if(DSEE_UNICORN_PROBE_ONLY OR DSEE_DYNARMIC_PROBE_ONLY)
  return()
endif()

target_include_directories(dsee_compat PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src)
target_link_libraries(dsee_compat PRIVATE dsee_support)
dsee_options(dsee_compat)

  target_sources(dsee_compat PRIVATE src/backend_native.cpp)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(dsee_compat PRIVATE _GNU_SOURCE)
    target_link_libraries(dsee_compat PRIVATE ${CMAKE_DL_LIBS})
  endif()
else()
  target_sources(dsee_compat PRIVATE src/backend_emulated.cpp)
  target_link_libraries(dsee_compat PRIVATE dsee_emulation)
endif()

target_include_directories(dsee_postprocess PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
dsee_options(dsee_postprocess)

target_include_directories(dsee-compat PRIVATE src)
target_link_libraries(dsee-compat PRIVATE dsee_compat dsee_postprocess)
dsee_options(dsee-compat)

  dsee_options(unit_tests)
  add_test(NAME unit COMMAND unit_tests)
  add_executable(api_test tests/api.cpp)
  target_include_directories(api_test PRIVATE src tools)
  target_link_libraries(api_test PRIVATE dsee_compat)
  if(WIN32)
    target_link_libraries(api_test PRIVATE psapi)

  dsee_options(api_test)
  add_test(NAME api_without_assets COMMAND api_test)
  add_executable(c_header_test tests/c_header.c)
  target_link_libraries(c_header_test PRIVATE dsee_compat)
  add_test(NAME c_header COMMAND c_header_test)
