cmake_minimum_required(VERSION 3.20)
project(dsee_compat VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
  message(FATAL_ERROR "The native runtime currently requires Apple Silicon macOS")
endif()
add_library(dsee_compat STATIC src/elf.cpp src/runtime.cpp)
target_include_directories(dsee_compat PUBLIC include PRIVATE src)
target_compile_options(dsee_compat PRIVATE -Wall -Wextra -Wpedantic)
add_executable(dsee-compat src/cli.cpp src/wav.cpp)
target_include_directories(dsee-compat PRIVATE src)
target_link_libraries(dsee-compat PRIVATE dsee_compat)
target_compile_options(dsee-compat PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(unit_tests tests/unit.cpp src/wav.cpp)
  target_include_directories(unit_tests PRIVATE src)
  target_link_libraries(unit_tests PRIVATE dsee_compat)
  add_test(NAME unit COMMAND unit_tests)
  add_executable(api_test tests/api.cpp)
  target_link_libraries(api_test PRIVATE dsee_compat)
  add_test(NAME api_without_assets COMMAND api_test)
  add_executable(c_header_test tests/c_header.c)
  target_link_libraries(c_header_test PRIVATE dsee_compat)
  add_test(NAME c_header COMMAND c_header_test)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME cli_without_assets COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py $<TARGET_FILE:dsee-compat>)
endif()
install(TARGETS dsee_compat dsee-compat ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(FILES include/dsee_compat.h DESTINATION include)
